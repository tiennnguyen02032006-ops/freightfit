const SOURCE_SHORT: Record<BlackBoxSource, string> = {
  gps: 'GPS',
  temp: 'Nhiệt độ',
  door: 'Cảm biến cửa',
  camera: 'Camera',
  engine: 'Telemetry xe',
  traffic: 'Dữ liệu giao thông',
  docs: 'Chứng từ',
  driver: 'Lời khai tài xế',
};

const TIERS: Record<'A' | 'B' | 'D', BlackBoxSource[]> = {
  A: ['gps', 'docs', 'driver'],
  B: ['gps', 'docs', 'driver', 'temp', 'engine'],
  D: BLACKBOX_SOURCES,
};

// Nhãn A/B/C cho 3 giả thuyết đứng đầu (đúng thứ tự result.hypotheses, đã xếp giảm dần theo độ tin
// cậy) — dùng ở cột "Bằng chứng" để người xem tự chọn xem chứng cứ ủng hộ/phản bác của giả thuyết
// nào, không chỉ mặc định xem giả thuyết đứng đầu.
const HYP_LETTER = ['A', 'B', 'C'] as const;

const pct = (p: number) => (p >= 0.995 ? '≥99%' : p < 0.005 ? '<1%' : `${Math.round(p * 100)}%`);
const pct0 = (p: number) => `${Math.round(p * 100)}%`;

interface Shade {
  a: number;
  b: number;
  className: string;
}

interface LineChartProps {
  label: string;
  height?: number;
  xmax: number;
  ymin: number;
  ymax: number;
  yticks: number[];
  points: Array<[number, number]>;
  shades?: Shade[];
  hline?: { value: number; label: string };
  lineClass?: string;
  // false = bỏ nhãn giờ ở trục dưới — dùng cho biểu đồ XẾP CHỒNG phía TRÊN 1 biểu đồ khác cùng
  // trục x (chỉ biểu đồ DƯỚI CÙNG mới cần hiện nhãn giờ, tránh lặp lại 2 lần cho cùng 1 trục thời
  // gian chung — xem JourneyChart).
  showXAxis?: boolean;
}

// Vẽ theo đúng thang đo: mọi tick/nhãn/điểm đều đi qua cùng 1 hàm x()/y().
function LineChart({
  label,
  height = 170,
  xmax,
  ymin,
  ymax,
  yticks,
  points,
  shades = [],
  hline,
  lineClass = 'bb-line',
  showXAxis = true,
}: LineChartProps) {
  const W = 460, H = height, L = 34, R = 8, T = 8, B = showXAxis ? 22 : 8;
  const x = (t: number) => L + (t / xmax) * (W - L - R);
  const y = (v: number) => H - B - ((v - ymin) / (ymax - ymin)) * (H - B - T);
  const xticks: number[] = [];
  if (showXAxis) {
    for (let t = 0; t <= xmax; t += 60) xticks.push(t);
  }
  return (
    <svg viewBox={`0 0 ${W} ${H}`} role="img" aria-label={label} className="bb-chart">
      {yticks.map((v) => (
        <g key={`y${v}`}>
          <line className="bb-grid" x1={L} x2={W - R} y1={y(v)} y2={y(v)} />
          <text x={L - 5} y={y(v) + 3} textAnchor="end">{v}</text>
        </g>
      ))}
      {xticks.map((t) => (
        <text key={`x${t}`} x={x(t)} y={H - 6} textAnchor={t + 60 > xmax ? 'end' : 'middle'}>{clockLabel(t)}</text>
      ))}
      {shades.map((s, i) => (
        <rect
          key={`s${i}`}
          className={s.className}
          x={x(s.a)}
          y={T}
          width={Math.max(1, x(Math.min(s.b, xmax)) - x(s.a))}
          height={H - B - T}
        />
      ))}
      {hline && (
        <g>
          <line className="bb-hline" x1={L} x2={W - R} y1={y(hline.value)} y2={y(hline.value)} />
          <text className="bb-hline-label" x={W - R} y={y(hline.value) - 3} textAnchor="end">{hline.label}</text>
        </g>
      )}
      <polyline
        className={lineClass}
        points={points
          .map(([t, v]) => `${x(t).toFixed(1)},${y(Math.min(ymax, Math.max(ymin, v))).toFixed(1)}`)
          .join(' ')}
      />
    </svg>
  );
}

interface JourneyChartProps {
  rec: BlackBoxCaseRecord;
  info: BlackBoxInfo;
  timeline: Array<{ t: number; text: string; level: 'info' | 'warn' | 'bad' }>;
}

/** Tốc độ + nhiệt độ dùng CHUNG 1 trục thời gian (2 sparkline xếp chồng, cùng xmax/cùng chiều rộng
 * nên các mốc giờ thẳng hàng theo chiều dọc) + 1 hàng mốc sự kiện dựng lại từ timeline bên dưới. */
function JourneyChart({ rec, info, timeline }: JourneyChartProps) {
  const xmax = Math.ceil(rec.meta.ata / 60) * 60;

  const speedShades: Shade[] = [
    ...info.stops.map((w) => ({ a: w[0], b: w[1], className: 'bb-shade-stop' })),
    ...info.crawls.map((w) => ({ a: w[0], b: w[1], className: 'bb-shade-crawl' })),
  ];
  if (info.offroute) speedShades.push({ a: info.offroute[0], b: info.offroute[1], className: 'bb-shade-off' });

  const maxTemp = rec.temp ? Math.max(...rec.temp.map((p) => p.T)) : null;
  const tempYmax = maxTemp !== null ? Math.max(12, Math.ceil(maxTemp / 4) * 4) : 12;
  const tempShades: Shade[] = info.doorOpen.map((w) => ({ a: w[0], b: w[1], className: 'bb-shade-off' }));

  return (
    <div className="bb-journey">
      <div className="bb-journey-row">
        <span className="bb-journey-label">Tốc độ</span>
        {rec.gps ? (
          <LineChart
            label="Tốc độ theo thời gian"
            height={90}
            xmax={xmax}
            ymin={0}
            ymax={80}
            yticks={[0, 40, 80]}
            points={rec.gps.map((p) => [p.t, p.v])}
            shades={speedShades}
            showXAxis={false}
          />
        ) : (
          <div className="bb-na bb-na-sm">Nguồn GPS đang tắt</div>
        )}
      </div>
      <div className="bb-journey-row">
        <span className="bb-journey-label">Nhiệt độ</span>
        {rec.temp ? (
          <LineChart
            label="Nhiệt độ theo thời gian"
            height={90}
            xmax={xmax}
            ymin={0}
            ymax={tempYmax}
            yticks={[0, tempYmax / 2, tempYmax]}
            points={rec.temp.map((p) => [p.t, p.T])}
            shades={tempShades}
            hline={{ value: rec.meta.setpoint + 1.5, label: 'ngưỡng' }}
            lineClass="bb-line-hot"
          />
        ) : (
          <div className="bb-na bb-na-sm">Cảm biến nhiệt độ đang tắt</div>
        )}
      </div>
      <div className="bb-legend">
        <span><i className="bb-shade-stop" />dừng xe</span>
        <span><i className="bb-shade-crawl" />bò chậm</span>
        <span><i className="bb-shade-off" />lệch tuyến / cửa mở</span>
        <span>tốc độ: km/h · nhiệt độ: °C</span>
      </div>
      <div className="bb-events-row">
        <span className="bb-journey-label">Sự kiện</span>
        <div className="bb-events-list">
          {timeline.length > 0 ? (
            timeline.map((e, i) => (
              <span key={`${e.t}-${i}`} className={`bb-event bb-tl-${e.level}`}>
                ● {clockLabel(e.t)} {e.text}
              </span>
            ))
          ) : (
            <span className="bb-muted">Chưa có mốc nào dựng lại được.</span>
          )}
        </div>
      </div>
    </div>
  );
}

export function BlackBoxPanel() {
  const [caseIndex, setCaseIndex] = useState(0);
  const [enabled, setEnabled] = useState<Record<BlackBoxSource, boolean>>(
    () => Object.fromEntries(BLACKBOX_SOURCES.map((s) => [s, true])) as Record<BlackBoxSource, boolean>,
  );
  const [reveal, setReveal] = useState(false);
  // Giả thuyết đang chọn để xem chứng cứ (0 = A = đứng đầu) — cột "Bằng chứng", xem JSX bên dưới.
  const [selectedHyp, setSelectedHyp] = useState(0);
  const [statsOpen, setStatsOpen] = useState(false);

  const fullCase = BLACKBOX_CASES[caseIndex];
  const result = useMemo(() => investigate(maskSources(fullCase, enabled)), [fullCase, enabled]);
  const maskedCase = useMemo(() => maskSources(fullCase, enabled), [fullCase, enabled]);

  // Đổi ca -> giả thuyết A (đứng đầu) là lựa chọn mặc định hợp lý nhất cho ca MỚI, không giữ lại
  // lựa chọn B/C của ca trước (dễ trỏ nhầm sang giả thuyết không tồn tại/không liên quan). Đặt
  // TRỰC TIẾP ở 2 nơi duy nhất đổi caseIndex (chọn tay + ngẫu nhiên) thay vì 1 effect riêng lắng
  // nghe caseIndex, để tránh setState đồng bộ trong effect.
  const changeCase = (index: number) => {
    setCaseIndex(index);
    setSelectedHyp(0);
  };

  const applyTier = (tier: 'A' | 'B' | 'D') => {
    const keep = TIERS[tier];
    setEnabled(Object.fromEntries(BLACKBOX_SOURCES.map((s) => [s, keep.includes(s)])) as Record<BlackBoxSource, boolean>);
  };
  const pickRandom = () => {
    if (BLACKBOX_CASES.length < 2) return;
    let n = caseIndex;
    while (n === caseIndex) n = Math.floor(Math.random() * BLACKBOX_CASES.length);
    changeCase(n);
  };

  const truth = fullCase.truth.causes;
  const hit = truth.includes(result.topCause);
  const maxTemp = fullCase.temp ? Math.max(...fullCase.temp.map((p) => p.T)) : null;
  const claims = fullCase.driver
    ? [
        !fullCase.driver.stop && 'không dừng xe ngoài kế hoạch',
        !fullCase.driver.dev && 'không đi lệch tuyến',
        !fullCase.driver.door && 'không mở cửa thùng',
      ].filter(Boolean).join(', ')
    : '';

  const enabledCount = BLACKBOX_SOURCES.filter((s) => enabled[s]).length;
  const top = result.hypotheses[0];
  const rest = result.hypotheses.slice(1);
  const activeHyp = result.hypotheses[Math.min(selectedHyp, result.hypotheses.length - 1)];

  return (
    <div className="bb-root">
      <p className="bb-banner">
        Dữ liệu MÔ PHỎNG. Kết quả là suy luận từ bằng chứng có độ tin cậy ước lượng, chưa phải kết luận chắc chắn.
      </p>

      <section className="panel">
        <div className="bb-header-row">
          <label className="bb-field-label bb-sr-only" htmlFor="bb-case">Ca điều tra</label>
          <select id="bb-case" className="bb-select bb-select-inline" value={caseIndex} onChange={(e) => changeCase(Number(e.target.value))}>
            {BLACKBOX_CASES.map((c, i) => (
              <option key={c.tripId} value={i}>
                {i === 0 ? '★ ' : ''}{c.tripId} · trễ {Math.round(c.meta.delay)}′{c.truth.causes.length > 1 ? ' · nhiều nguyên nhân' : ''}
              </option>
            ))}
          </select>
          <button type="button" className="bb-btn bb-btn-primary" onClick={pickRandom}>Ca ngẫu nhiên</button>
          <label className="bb-check bb-check-inline">
            <input type="checkbox" checked={reveal} onChange={(e) => setReveal(e.target.checked)} />
            Hiện đáp án thật để chấm
          </label>
        </div>
        <div className="bb-chips">
          <span className={`bb-chip ${fullCase.meta.delay > 20 ? 'bb-chip-bad' : ''}`}>Trễ {Math.round(fullCase.meta.delay)} phút</span>
          {maxTemp !== null && (
            <span className={`bb-chip ${maxTemp - fullCase.meta.setpoint > 1.5 ? 'bb-chip-bad' : ''}`}>
              Nhiệt độ tối đa {maxTemp.toFixed(1)}°C
            </span>
          )}
          {fullCase.docs && fullCase.docs.loaded !== fullCase.docs.delivered && (
            <span className="bb-chip bb-chip-warn">Thiếu {fullCase.docs.loaded - fullCase.docs.delivered} kiện</span>
          )}
          {fullCase.driver && <span className="bb-chip">Tài xế khai: {claims || 'không nêu bất thường'}</span>}
        </div>
        {reveal && (
          <p className={`bb-verdict ${hit ? 'bb-good' : 'bb-miss'}`}>
            Đáp án thật: <b>{truth.map((c) => CAUSE_LABEL[c]).join(' + ')}</b>. Engine xếp đầu:{' '}
            <b>{CAUSE_LABEL[result.topCause]}</b> → {hit ? 'đúng' : 'sai'}
            {truth.length > 1 ? ' (ca nhiều nguyên nhân, engine chỉ chọn một)' : ''}.
          </p>
        )}
      </section>

      <section className="panel">
        <h2>Kết luận</h2>
        {top && (
          <div className="bb-hyp bb-hyp-top">
            <div className="bb-hyp-head">
              <b>{CAUSE_LABEL[top.cause]}</b>
              <span className="bb-conf">{pct(top.confidence)}</span>
            </div>
            <div className="bb-bar"><i style={{ width: `${(top.confidence * 100).toFixed(1)}%` }} /></div>
          </div>
        )}
        {rest.length > 0 && (
          <ul className="bb-hyp-compact">
            {rest.map((h, i) => (
              <li key={h.cause}>
                <span className="bb-muted">{i + 2}:</span> {CAUSE_LABEL[h.cause]}{' '}
                <span className="bb-conf">{pct(h.confidence)}</span>
              </li>
            ))}
          </ul>
        )}
      </section>

      <section className="panel">
        <h2>Nguồn dữ liệu có sẵn</h2>
        <div className="bb-src-row">
          {BLACKBOX_SOURCES.map((s) => (
            <label key={s} className="bb-src bb-src-chip">
              <input
                type="checkbox"
                checked={enabled[s]}
                onChange={(e) => setEnabled((prev) => ({ ...prev, [s]: e.target.checked }))}
              />
              {SOURCE_SHORT[s]}
            </label>
          ))}
        </div>
        <div className="bb-row">
          <button type="button" className="bb-btn" onClick={() => applyTier('A')}>Gói A</button>
          <button type="button" className="bb-btn" onClick={() => applyTier('B')}>Gói B</button>
          <button type="button" className="bb-btn" onClick={() => applyTier('D')}>Đầy đủ</button>
          <span className="bb-counter">Đang dùng {enabledCount}/{BLACKBOX_SOURCES.length} nguồn</span>
        </div>
        <p className="bb-note">A: GPS, chứng từ, lời khai. B: thêm nhiệt độ và telemetry xe.</p>
      </section>

      <section className="panel">
        <h2>Hành trình (một trục giờ chung)</h2>
        <JourneyChart rec={maskedCase} info={result.info} timeline={result.timeline} />
      </section>

      <div className="bb-cols3">
        <section className="panel">
          <h2>Mâu thuẫn</h2>
          {result.info.contradictions.length > 0 ? (
            result.info.contradictions.map(([claim, evidence]) => (
              <div key={claim} className="bb-contra">
                <span>{claim}</span><em>≠</em><span>{evidence}</span>
              </div>
            ))
          ) : (
            <p className="bb-none">{enabled.driver ? 'Không phát hiện mâu thuẫn.' : 'Chưa có lời khai để đối chiếu.'}</p>
          )}
        </section>

        <section className="panel">
          <h2>Bằng chứng</h2>
          <div className="bb-row bb-row-tight">
            {result.hypotheses.map((h, i) => (
              <button
                key={h.cause}
                type="button"
                className={`bb-btn bb-ev-btn${i === selectedHyp ? ' is-active' : ''}`}
                onClick={() => setSelectedHyp(i)}
                title={CAUSE_LABEL[h.cause]}
              >
                {HYP_LETTER[i] ?? i + 1}
              </button>
            ))}
          </div>
          {activeHyp && (
            <>
              <p className="bb-note bb-ev-current">
                Giả thuyết {HYP_LETTER[selectedHyp] ?? selectedHyp + 1}: <b>{CAUSE_LABEL[activeHyp.cause]}</b> ({pct(activeHyp.confidence)})
              </p>
              <ul className="bb-evidence">
                {activeHyp.support.slice(0, 5).map(([f, v]) => (
                  <li key={`s-${f}`} className="bb-yes">✓ {valueLabel(f, v)}</li>
                ))}
                {activeHyp.refute.slice(0, 3).map(([f, v]) => (
                  <li key={`r-${f}`} className="bb-no">✗ {valueLabel(f, v)}</li>
                ))}
                {activeHyp.support.length === 0 && activeHyp.refute.length === 0 && (
                  <li className="bb-muted">Chưa có chứng cứ nào ủng hộ/phản bác rõ rệt.</li>
                )}
              </ul>
            </>
          )}
        </section>

        <section className="panel">
          <h2>Bước tiếp theo</h2>
          <p className="bb-note bb-h2-gap-sm">Cần thêm:</p>
          <ul className="bb-list">
            {result.requests.length > 0 ? (
              result.requests.map(([s, g]) => (
                <li key={s}>{SOURCE_LABEL[s]} <span className="bb-muted">(giá trị phân biệt {g.toFixed(1)})</span></li>
              ))
            ) : (
              <li className="bb-muted">Bằng chứng hiện có đã đủ phân biệt.</li>
            )}
          </ul>
          <p className="bb-note bb-h2-gap-sm">Hành động đề xuất:</p>
          <ul className="bb-list">
            {CAUSE_ACTIONS[result.topCause].map((a) => <li key={a}>{a}</li>)}
          </ul>
        </section>
      </div>

      <section className="panel bb-collapsible">
        <button type="button" className="bb-collapsible-head" onClick={() => setStatsOpen((v) => !v)} aria-expanded={statsOpen}>
          <span>{statsOpen ? '▾' : '▸'} Độ tin cậy của prototype này</span>
        </button>
        {statsOpen && (
          <>
            <table className="bb-table">
              <tbody>
                {BLACKBOX_STATS.tiers.map(([name, v]) => (
                  <tr key={name}><td>{name}</td><td className="bb-num">{pct0(v)}</td></tr>
                ))}
                <tr><td>Sự cố có 2 nguyên nhân cùng lúc (đoán đúng cả cặp)</td><td className="bb-num">{pct0(BLACKBOX_STATS.mixedExactPair)}</td></tr>
                <tr><td>Nhiễu nặng và yếu tố lạ</td><td className="bb-num">{pct0(BLACKBOX_STATS.worstCase)}</td></tr>
              </tbody>
            </table>
            <p className="bb-note">
              Tỉ lệ đoán đúng nguyên nhân xếp đầu trên 1.800 ca mô phỏng mỗi điều kiện (ngẫu nhiên: {pct0(BLACKBOX_STATS.chance)}).
              Dữ liệu và mô hình cùng do người làm dự án dựng, nên đây là mức trần, chưa phải kết quả thực tế.
            </p>
          </>
        )}
      </section>
    </div>
  );
}
